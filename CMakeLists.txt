cmake_minimum_required(VERSION 3.13)
if(NOT DEFINED PICO_SDK_PATH)
  message(FATAL_ERROR "Configure with -DPICO_SDK_PATH=/path/to/pico-sdk")
endif()
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(cansat C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()

function(cansat_target name)
  add_executable(${name} pico/firmware.cpp)
  target_include_directories(${name} PRIVATE pico)
  target_link_libraries(${name} pico_stdlib hardware_i2c hardware_spi)
  pico_enable_stdio_usb(${name} 1)
  pico_enable_stdio_uart(${name} 0)
  pico_add_extra_outputs(${name})
endfunction()

cansat_target(cansat_flight)
cansat_target(cansat_ground)
target_compile_definitions(cansat_ground PRIVATE GROUND_STATION=1)
